cmake_minimum_required(VERSION 3.22.1)
project(familiar_runtime)
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_MOBILE_ROOT}/bin/${ANDROID_ABI}/libnode.so")
add_library(familiar_runtime SHARED runtime.cpp)
target_compile_features(familiar_runtime PRIVATE cxx_std_20)
target_include_directories(familiar_runtime PRIVATE "${NODE_MOBILE_ROOT}/include/node")
target_link_libraries(familiar_runtime node log)
target_link_options(familiar_runtime PRIVATE "-Wl,-z,max-page-size=16384")

add_library(familiar_workspace SHARED "${CMAKE_CURRENT_LIST_DIR}/../../../../../feishu-android/wasm/workspace.cpp")
target_compile_features(familiar_workspace PRIVATE cxx_std_20)
target_compile_definitions(familiar_workspace PRIVATE NODE_GYP_MODULE_NAME=familiar_workspace)
target_include_directories(familiar_workspace PRIVATE "${NODE_MOBILE_ROOT}/include/node")
target_link_libraries(familiar_workspace node log)
target_link_options(familiar_workspace PRIVATE "-Wl,-z,max-page-size=16384")
